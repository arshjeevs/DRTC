% --- the report's own macros -----------------------------------------------------------------------------------------
% Results: report/scripts/build.py writes generated/numbers.tex with \setR{key}{value}; the text writes \R{key}.
% A key that the results do not give yet prints "(pending)", so the text can be written before the result exists.
\makeatletter
\newcommand{\setR}[2]{\expandafter\def\csname R@#1\endcsname{#2}}
\newcommand{\R}[1]{\ifcsname R@#1\endcsname\csname R@#1\endcsname\else\pending\fi}
\makeatother
\newcommand{\pending}{\textit{(pending)}}
\InputIfFileExists{generated/numbers.tex}{}{}

% A framed box standing in for something not made yet (a photo, a figure, a table).
\newcommand{\placeholderbox}[3][0.8\textwidth]{%
  \fbox{\parbox[c][#2][c]{#1}{\centering\small\itshape #3}}}
% Generated tables and figures (from the results); "pending" until the build script has made them.
\newcommand{\gentable}[1]{\IfFileExists{generated/#1.tex}{\input{generated/#1.tex}}%
  {\placeholderbox[0.9\textwidth]{3cm}{Table pending: \texttt{generated/\detokenize{#1}.tex} is made when its results exist.}}}
\newcommand{\genfigure}[2][\textwidth]{\IfFileExists{generated/#2.pdf}{\includegraphics[width=#1]{generated/#2.pdf}}%
  {\placeholderbox[0.9\textwidth]{5cm}{Figure pending: \texttt{generated/\detokenize{#2}.pdf} is made when its results exist.}}}
% Photographs and video stills (assets/): the photo when the file exists, else a box saying what to put there.
\newcommand{\photo}[3][0.8\textwidth]{\IfFileExists{assets/#2}{\includegraphics[width=#1]{assets/#2}}%
  {\placeholderbox[#1]{5.5cm}{Photo to add: \texttt{assets/\detokenize{#2}}\\[4pt] #3}}}
