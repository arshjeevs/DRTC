% --- page and type ---------------------------------------------------------------------------------------------------
% Layout of the NIT Calicut final-year project report (Final-Year-Project-Report.docx, 2026-11-26): A4, text about
% 1 inch from each side, Times New Roman 12 pt, justified, line spacing 1.15, no running header, page number centred.

% --- headings, headers ------------------------------------------------------------------------------------------------
\usepackage{titlesec}
% Chapters: "CHAPTER 1" over the title, both centred, bold, 16 pt, upper case. Sections 14 pt bold upper case (1.1),
% subsections 12 pt bold upper case (1.1.1), subsubsections 12 pt bold (1.1.1.1). Every heading is left in the text's
% own case and upper-cased here, so the contents list keeps normal case.
\titleformat{\chapter}[display]{\normalfont\fontsize{16}{20}\selectfont\bfseries\filcenter}%
  {\MakeUppercase{\chaptertitlename}\ \thechapter}{6pt}{\MakeUppercase}
\titleformat{name=\chapter,numberless}{\normalfont\fontsize{16}{20}\selectfont\bfseries\filcenter}{}{0pt}%
  {\MakeUppercase}
\titlespacing*{\chapter}{0pt}{0pt}{18pt}
\titleformat{\section}{\normalfont\fontsize{14}{17}\selectfont\bfseries}{\thesection}{0.8em}{\MakeUppercase}
\titleformat{\subsection}{\normalfont\normalsize\bfseries}{\thesubsection}{0.8em}{\MakeUppercase}
\titleformat{\subsubsection}{\normalfont\normalsize\bfseries}{\thesubsubsection}{0.8em}{}
\titlespacing*{\section}{0pt}{14pt}{6pt}
\titlespacing*{\subsection}{0pt}{12pt}{4pt}
\titlespacing*{\subsubsection}{0pt}{10pt}{2pt}
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{2}
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\thepage}\renewcommand{\headrulewidth}{0pt}}

% --- the report's own macros -----------------------------------------------------------------------------------------
% Results: report/scripts/build.py writes generated/numbers.tex with \setR{key}{value}; the text writes \R{key}.
% A key that the results do not give yet prints "(pending)", so the text can be written before the result exists.
\makeatletter
\newcommand{\setR}[2]{\expandafter\def\csname R@#1\endcsname{#2}}
\newcommand{\R}[1]{\ifcsname R@#1\endcsname\csname R@#1\endcsname\else\pending\fi}
\makeatother
\newcommand{\pending}{\textit{(pending)}}
\InputIfFileExists{generated/numbers.tex}{}{}

% A framed box standing in for something not made yet (a photo, a figure, a table).
\newcommand{\placeholderbox}[3][0.8\textwidth]{%
  \fbox{\parbox[c][#2][c]{#1}{\centering\small\itshape #3}}}
% Generated tables and figures (from the results); "pending" until the build script has made them.
\newcommand{\gentable}[1]{\IfFileExists{generated/#1.tex}{\input{generated/#1.tex}}%
  {\placeholderbox[0.9\textwidth]{3cm}{Table pending: \texttt{generated/\detokenize{#1}.tex} is made when its results exist.}}}
\newcommand{\genfigure}[2][\textwidth]{\IfFileExists{generated/#2.pdf}{\includegraphics[width=#1]{generated/#2.pdf}}%
  {\placeholderbox[0.9\textwidth]{5cm}{Figure pending: \texttt{generated/\detokenize{#2}.pdf} is made when its results exist.}}}
% Photographs and video stills (assets/): the photo when the file exists, else a box saying what to put there.
\newcommand{\photo}[3][0.8\textwidth]{\IfFileExists{assets/#2}{\includegraphics[width=#1]{assets/#2}}%
  {\placeholderbox[#1]{5.5cm}{Photo to add: \texttt{assets/\detokenize{#2}}\\[4pt] #3}}}
