\input{figures/colors}
\begin{tikzpicture}[font=\footnotesize, node distance=6mm and 9mm,
  st/.style={draw=greenline, thick, rounded corners=3pt, fill=white, align=center, text width=35mm, minimum height=10mm},
  dec/.style={draw=ink2, thick, diamond, aspect=2.4, fill=netfill, align=center, inner sep=1pt, text width=22mm},
  fix/.style={circle, fill=purpleline, text=white, font=\scriptsize\bfseries, inner sep=1.2pt},
  arr/.style={-{Latex[length=2mm]}, thick, ink2}, lab/.style={font=\scriptsize, text=ink2}]
\node[st] (trig) {at most $H-s_{\min}$ actions left: send the freshest observation};
\node[st, below=of trig] (cool) {cooldown from the latency estimate (Jacobson--Karels)};
\node[dec, below=of cool] (rep) {reply before the cooldown ends?};
\node[st, right=14mm of rep] (merge) {merge (last write wins); update the latency estimate};
\node[st, below=of rep] (re) {give the request up; send a fresh observation};
\node[st, right=14mm of re] (hold) {queue empty meanwhile: hold the arm (the run ends only if no reply comes within LeRobot's 10~s deadline)};
\draw[arr] (trig) -- (cool); \draw[arr] (cool) -- (rep);
\draw[arr] (rep) -- node[lab, above]{yes} (merge); \draw[arr] (rep) -- node[lab, left]{no} (re);
\draw[arr] (re.west) -- ++(-6mm,0) |- (cool.west);
\draw[arr, dashed] (re) -- (hold);
\draw[arr] (merge.north) |- (trig.east);
% fixes
\node[fix] at ([xshift=-1mm, yshift=1mm]cool.north east) {1};
\node[fix] at ([xshift=-5.5mm, yshift=1mm]cool.north east) {2};
\node[fix] at ([xshift=-10mm, yshift=1mm]cool.north east) {3};
\node[fix] at ([xshift=1mm, yshift=1mm]trig.north west) {4};
\node[draw=purpleline, rounded corners=3pt, align=left, text width=95mm, font=\scriptsize, anchor=north west] at
  ([yshift=-6mm]re.south west) {\textbf{Our fixes}\\[1pt]
  \textbf{1} the cooldown is never shorter than the measured latency (no $H/2$ cap)\\
  \textbf{2} it also covers the slowest of the last 10 replies\\
  \textbf{3} before the first reply, wait up to the start-up timeout\\
  \textbf{4} the requested inference delay is capped at the server's prediction steps};
\end{tikzpicture}
