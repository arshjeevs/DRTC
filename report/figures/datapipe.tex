% D4.1 The data pipeline: teleoperation -> recording -> review -> trimming -> split and version -> Hub.
\input{figures/colors}
\begin{tikzpicture}[font=\small, node distance=4mm and 5.5mm,
  st/.style={draw=#1, thick, rounded corners=3pt, fill=white, align=center, text width=24mm, minimum height=21mm,
             font=\footnotesize},
  arr/.style={-{Latex[length=2.2mm]}, very thick, ink2}]
\node[st=robotline] (tele) {{\bfseries Teleoperation}\\[2pt] a person moves the leader arm; the follower copies it};
\node[st=robotline, right=of tele] (rec) {{\bfseries Recording}\\[2pt] both cameras and the joint states at 30~fps,
  one episode per demonstration};
\node[st=robotline, right=of rec] (rev) {{\bfseries Review}\\[2pt] bad episodes deleted and recorded again};
\node[st=greenline, right=of rev] (trim) {{\bfseries Trimming}\\[2pt] idle start and end cut automatically; raw copy
  kept};
\node[st=purpleline, right=of trim] (split) {{\bfseries Split and version}\\[2pt] seeded held-out episodes; pinned
  Hub revision};
\foreach \a/\b in {tele/rec, rec/rev, rev/trim, trim/split} \draw[arr] (\a) -- (\b);
\draw[arr] (rev.north) -- ++(0,5mm) -| node[pos=0.25, above, font=\scriptsize]{re-record} (rec.north);
\node[below=3mm of split, font=\scriptsize, text=ink2, align=center] {Hugging Face Hub;\\ \texttt{recorded.yaml} in git};
\end{tikzpicture}
