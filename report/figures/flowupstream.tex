% D6.2 LeRobot's client: what happens when a reply is late or lost (as shipped / patient).
\input{figures/colors}
\begin{tikzpicture}[font=\footnotesize, node distance=6mm and 9mm,
  st/.style={draw=robotline, thick, rounded corners=3pt, fill=white, align=center, text width=33mm, minimum height=10mm},
  dec/.style={draw=ink2, thick, diamond, aspect=2.4, fill=netfill, align=center, inner sep=1pt, text width=22mm},
  bad/.style={draw=serverline, thick, rounded corners=3pt, fill=serverfill, align=center, text width=33mm,
              minimum height=10mm},
  arr/.style={-{Latex[length=2mm]}, thick, ink2}, lab/.style={font=\scriptsize, text=ink2}]
\node[st] (req) {queue below the refill threshold: send a request};
\node[dec, below=of req] (rep) {reply within the request deadline?};
\node[st, right=14mm of rep] (merge) {merge the chunk; keep executing};
\node[dec, below=of rep] (dry) {queue empty?};
\node[st, right=14mm of dry] (wait) {keep executing the queue and check again};
\node[st, below=of dry] (forget) {hold the arm; forget the request in flight; send one fresh request};
\node[dec, below=of forget] (lost) {that reply arrives?};
\node[st, right=14mm of lost] (resume) {resume};
\node[bad, below=10mm of lost, xshift=-24mm] (ship) {\textbf{as shipped:} the run ends after 1~s without actions
  (starvation grace) or 5~s without a reply};
\node[bad, below=10mm of lost, xshift=24mm] (pat) {\textbf{patient:} holds the arm and waits to the end of the trial;
  no further request is sent};
\draw[arr] (req) -- (rep); \draw[arr] (rep) -- node[lab, above]{yes} (merge);
\draw[arr] (rep) -- node[lab, left]{not yet} (dry); \draw[arr] (dry) -- node[lab, above]{no} (wait);
\draw[arr] (dry) -- node[lab, left]{yes} (forget); \draw[arr] (forget) -- (lost);
\draw[arr] (lost) -- node[lab, above]{yes} (resume);
\draw[arr] (lost.south) -- node[lab, left, pos=0.3]{no (lost)} ++(0,-4mm) -| (ship.north);
\draw[arr] (lost.south) ++(0,-4mm) -| (pat.north);
\draw[arr] (merge.north) |- (req.east);
\end{tikzpicture}
